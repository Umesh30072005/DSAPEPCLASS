public static long countPerfectSquaresInRange(long L, long R) {

    long start = (long) Math.ceil(Math.sqrt(L));
    long end = (long) Math.floor(Math.sqrt(R));
    if (start > end) {
        return 0;
    }
    return end - start + 1;
}

    // add your code here
    int left = 0;
    int right = ids.length-1;
    if(ids.length <= 0){
        return 0;
    }
    while(left<=right){
        int mid = left + (right - left)/2;
        if(ids[mid] == target ){
            return 1;
        }
        else if(ids[mid] < target){
            left = mid +1;
        }
        else if(ids[mid]>target){

Librearians inventory count :

public static int countOccurrences(int[] arr, int target) {

    int first = firstIndex(arr, target);
    if (first == -1) return 0;

    int last = lastIndex(arr, target);
    return last - first + 1;
}

// Find first occurrence
private static int firstIndex(int[] arr, int target) {
    int left = 0, right = arr.length - 1;
    int ans = -1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (arr[mid] == target) {
            ans = mid;
            right = mid - 1;   // move left
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return ans;
}

// Find last occurrence
private static int lastIndex(int[] arr, int target) {
    int left = 0, right = arr.length - 1;
    int ans = -1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (arr[mid] == target) {
            ans = mid;
            left = mid + 1;    // move right
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return ans;
}

public static int balancedTreasurePoint(int[] arr) {
    // add your code here
    int totalsum = 0;
    int leftsum = 0;
    for(int num : arr){
        totalsum += num;
    }
    for(int i = 0;i<arr.length;i++){
        int rightsum = totalsum - leftsum -arr[i];
        if(rightsum == leftsum){
            return i;
        }
        leftsum += arr[i];
    }
    return -1;
}

public ListNode addTwoNumbers(ListNode l1, ListNode l2, int k) {
    // add your code here
    ListNode dummy = new ListNode(0);
    ListNode curr = dummy;
    int carry = 0;
    while(l1 != null || l2 != null || carry != 0){
        int sum = carry;
        if(l1 != null){
            sum += l1.val;
            l1  = l1.next;
        }
        if(l2 != null){
            sum += l2.val;
            l2 = l2.next;
        }
        carry = sum/10;
        int digit = (sum %10);
        digit = (digit * k) % 10;
        curr.next = new ListNode(digit);
        curr = curr.next;
    }
    return dummy.next;
}

public static int matrixBorderSum(int[][] grid) {
    // add your code here
    int row  = grid.length;
    int col = grid[0].length;
    int top = 0;
    int bottom = row -1;
    int left = 0;
    int right = col -1;
    int sum = 0;
        for(int j = left;j<= right;j++){
            sum += grid[top][j];
        }
        for(int i = top+1; i<= bottom;i++){
            sum += grid[i][right];
        }
        if(top != bottom){
            for(int j = right -1;j>=left;j--){
                sum += grid[bottom][j];
            }
        }
        if(left != right){
            for(int i = bottom -1;i>top;i--){
                sum += grid[i][left];
            }
        }
        return sum;
}

public static boolean isPalindrome(Node head) {
    // add your code here
    if(head == null || head.next == null){
        return true;
    }
    Node slow = head;
    Node fast  = head;
    while(fast != null && fast.next != null){
        slow = slow.next;
        fast = fast.next.next;
    }
    Node prev = null;
    Node curr = slow;
    while(curr != null){
        Node next = curr.next;
        curr.next = prev;
        prev = curr;
        curr = next;
    }
    Node first = head;
    Node second = prev;
    while(second != null){
        if(first.data != second.data){
            return false;
        }
        first = first.next;
        second = second.next;
    }
    return true;
}

Insert node a kth position in  a linked list :

public static Node insertAtKthPosition(Node head, int data, int k) {
    // add your code here
    Node newnode = new Node(data);
    if(head == null){
        return newnode;
    }
    if(k == 1){
        newnode.next = head;
        head.prev = newnode;
        return newnode;
    }
    Node temp = head;
    int count = 1;
    while(temp.next != null && count < k-1){
        temp = temp.next;
        count++;
    }
    if(temp.next == null){
        temp.next = newnode;
        newnode.prev = temp;
        return head;
    }
    newnode.next = temp.next;
    newnode.prev = temp;
    temp.next.prev = newnode;
    temp.next = newnode;
    return head;
}

public static int scoreBrackets(String s) {
    // add your code here
    int score = 0;
    int depth =0;
    for(int i = 0;i<s.length();i++){
        if(s.charAt(i) == '(' ){
            depth++;
        }else{
            depth--;
            if(s.charAt(i-1) == '('){
                score += 1 <<depth;
            }
        }
    }
    return score;
}
